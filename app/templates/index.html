class AIProviderManager:
    """Dispatches requests dynamically to configured multimodal LLM providers"""
    
    def __init__(self, providers: Dict[str, BaseAIProvider]):
        self.providers = providers

    async def dispatch_stream(self, provider_name: str, prompt: str):
        provider = self.providers.get(provider_name)
        async for chunk in provider.stream_response(prompt):
            yield chunk
